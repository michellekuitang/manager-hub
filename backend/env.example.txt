# Modele de configuration du backend.
# Copier ce fichier en .env et renseigner les valeurs reelles.
# Le fichier .env ne doit jamais etre versionne.

# Port d'ecoute du serveur.
# En production, l'hebergeur impose souvent sa propre valeur.
PORT=5000

# Chaine de connexion MongoDB.
#   local          : mongodb://localhost:27017/managerhub
#   docker compose : mongodb://mongo:27017/managerhub
#   production     : mongodb+srv://UTILISATEUR:MOTDEPASSE@cluster.xxxxx.mongodb.net/managerhub
MONGO_URI=mongodb://localhost:27017/managerhub

# Secret de signature des jetons JWT.
# Doit etre long et aleatoire. Generer une valeur differente en production :
#   node -e "console.log(require('crypto').randomBytes(48).toString('hex'))"
JWT_SECRET=a_remplacer_par_une_valeur_aleatoire

# Duree de validite des jetons.
JWT_EXPIRES_IN=8h

# Cle d'API Groq pour le generateur de contenu.
# A obtenir sur https://console.groq.com
GROQ_API_KEY=gsk_a_remplacer

# Origines autorisees a appeler l'API depuis un navigateur.
# Plusieurs valeurs separees par des virgules.
#   local      : http://localhost:3000
#   production : https://mon-frontend.onrender.com
CORS_ORIGINS=http://localhost:3000

# Environnement d'execution : development ou production.
NODE_ENV=development
